if(NOT WIN32)
    message(FATAL_ERROR "The native RaySculpt desktop currently requires Windows.")
endif()
find_package(Qt6 6.6 REQUIRED COMPONENTS Widgets)
find_package(OpenCASCADE REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
find_package(LibArchive REQUIRED)
find_program(RUSTC_EXECUTABLE rustc REQUIRED)
if(BUILD_TESTING)
    add_executable(native_cad_tests "${PROJECT_SOURCE_DIR}/tests/native_cad_tests.cpp")
    target_compile_options(native_cad_tests PRIVATE /utf-8)
    target_compile_definitions(native_cad_tests PRIVATE NOMINMAX)
    target_link_libraries(native_cad_tests PRIVATE raysculpt_cad)
    qt_add_resources(native_cad_tests test_materials PREFIX "/" BASE "${PROJECT_SOURCE_DIR}/tools" FILES "${PROJECT_SOURCE_DIR}/tools/materials.json")
    add_test(NAME native_cad COMMAND native_cad_tests)
endif()
set(transport "${CMAKE_CURRENT_BINARY_DIR}/raysculpt_transport.lib")
add_custom_command(OUTPUT "${transport}"
    COMMAND "${RUSTC_EXECUTABLE}" --edition=2024 --crate-type=staticlib -C opt-level=3
        "${CMAKE_CURRENT_SOURCE_DIR}/../transport/lib.rs" -o "${transport}"
    DEPENDS ../transport/lib.rs ../transport/numerics.rs VERBATIM)
add_custom_target(native_transport DEPENDS "${transport}")
if(BUILD_TESTING)
    set(transport_tests "${CMAKE_CURRENT_BINARY_DIR}/native_transport_tests.exe")
    add_custom_command(OUTPUT "${transport_tests}"
        COMMAND "${RUSTC_EXECUTABLE}" --edition=2024 --test -C opt-level=3
            "${CMAKE_CURRENT_SOURCE_DIR}/../transport/lib.rs" -o "${transport_tests}"
        DEPENDS ../transport/lib.rs ../transport/numerics.rs VERBATIM)
    add_custom_target(build_transport_tests ALL DEPENDS "${transport_tests}")
    add_test(NAME native_transport COMMAND "${transport_tests}")
endif()
add_library(raysculpt_cad STATIC inspection.cpp topology.cpp geometry.cpp document.cpp cad_body.cpp export.cpp)
target_include_directories(raysculpt_cad PUBLIC ${OpenCASCADE_INCLUDE_DIR})
target_compile_options(raysculpt_cad PRIVATE /utf-8 /bigobj)
target_compile_definitions(raysculpt_cad PRIVATE NOMINMAX)
target_link_libraries(raysculpt_cad PUBLIC Qt6::Gui caustic_core LibArchive::LibArchive
    TKernel TKMath TKBRep TKG2d TKG3d TKGeomBase TKGeomAlgo TKTopAlgo TKPrim TKBO TKBool TKMesh TKShHealing TKDESTEP TKDESTL)
target_link_libraries(caustic_cli PRIVATE raysculpt_cad)
target_compile_definitions(caustic_cli PRIVATE RAYSCULPT_NATIVE_EXPORT NOMINMAX)
target_compile_options(caustic_cli PRIVATE /utf-8)
add_library(raysculpt_photo STATIC jobs.cpp spectral_filter.cpp verify_solid.cpp verify_facets.cpp)
add_dependencies(raysculpt_photo native_transport)
target_include_directories(raysculpt_photo PRIVATE ${OpenCV_INCLUDE_DIRS})
target_compile_options(raysculpt_photo PRIVATE /utf-8)
target_compile_definitions(raysculpt_photo PRIVATE NOMINMAX)
target_link_libraries(raysculpt_photo PUBLIC raysculpt_cad ${OpenCV_LIBS} ${transport}
    advapi32 bcrypt ntdll userenv ws2_32 synchronization)
target_link_libraries(caustic_cli PRIVATE raysculpt_photo)
qt_add_resources(caustic_cli cli_materials PREFIX "/" BASE "${PROJECT_SOURCE_DIR}/tools" FILES "${PROJECT_SOURCE_DIR}/tools/materials.json")
qt_add_executable(raysculpt WIN32 main.cpp window.cpp properties.cpp commands.cpp optical_ui.cpp view.cpp
    ../photo_trace.cpp ../image_file.cpp ../plate_design.cpp)
add_dependencies(raysculpt native_transport)
file(GLOB icons CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/tools/icons/*.svg")
qt_add_resources(raysculpt editor_resources PREFIX "/" BASE "${PROJECT_SOURCE_DIR}/tools"
    FILES "${PROJECT_SOURCE_DIR}/tools/materials.json" ${icons})
target_include_directories(raysculpt PRIVATE ${OpenCASCADE_INCLUDE_DIR} ${OpenCV_INCLUDE_DIRS})
target_compile_options(raysculpt PRIVATE /utf-8 /bigobj)
target_compile_definitions(raysculpt PRIVATE NOMINMAX)
target_link_libraries(raysculpt PRIVATE Qt6::Widgets raysculpt_photo TKService TKV3d TKOpenGl ole32 windowscodecs)
